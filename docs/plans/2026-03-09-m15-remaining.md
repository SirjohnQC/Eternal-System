### What to build

As the player's planet progresses through biology phases, the rendered surface gets a subtle color tint that shifts the visual feel:

| Phase | Tint effect |
|-------|-------------|
| microbial | Faint cyan/teal overlay on land (barely visible — life just beginning) |
| multicellular | Soft green tint on mid-height land areas |
| complex | Richer green; slight darkening of water |
| primitive | Warm earthy tones; slight orange/brown shift |
| intelligent | No extra tint — city lights already handle this |

This is implemented as a **post-process blend** on the `heightToColor` return value for the player's home planet (only when `this.star.isPlayerStar`).

### Step 1: Add `applyBioTint()` helper to `PlanetRenderer.ts`

Read PlanetRenderer.ts first to find `heightToColor` and understand the file structure. Then add this private method:

```typescript
private applyBioTint(
  color: [number, number, number],
  h: number,
  phase: string,
): [number, number, number] {
  // Only tint land areas (h > 0.45 roughly) — don't tint ocean
  const isLand = h > 0.45;
  if (!isLand) return color;

  let [r, g, b] = color;

  switch (phase) {
    case 'microbial':
      // Very faint cyan shimmer
      r = Math.round(r * 0.97);
      g = Math.min(255, Math.round(g * 1.00 + 6));
      b = Math.min(255, Math.round(b * 1.00 + 10));
      break;
    case 'multicellular':
      // Soft green tint
      r = Math.round(r * 0.95);
      g = Math.min(255, Math.round(g * 1.05 + 8));
      b = Math.round(b * 0.97);
      break;
    case 'complex':
      // Richer green
      r = Math.round(r * 0.92);
      g = Math.min(255, Math.round(g * 1.08 + 12));
      b = Math.round(b * 0.95);
      break;
    case 'primitive':
      // Warm earthy — slight desaturation toward brown
      r = Math.min(255, Math.round(r * 1.03 + 5));
      g = Math.min(255, Math.round(g * 1.01 + 3));
      b = Math.round(b * 0.94);
      break;
    default:
      // 'intelligent' or unknown — no tint
      break;
  }

  return [
    Math.max(0, Math.min(255, r)),
    Math.max(0, Math.min(255, g)),
    Math.max(0, Math.min(255, b)),
  ];
}
```

### Step 4: Add `spawnCosmicSignal()` and `updateCosmicSignals()` methods

```typescript
private spawnCosmicSignal(star: StarBody): void {
  if (!this.isStarKnownToPlayer(star)) return;
  const msgs = BigBangEngine.SIGNAL_MESSAGES;
  const raw  = msgs[this.rng.nextInt(0, msgs.length - 1)];
  const message = raw.replace('{civ}', star.civName).replace('{dist}', String(this.rng.nextInt(4, 180)));
  this.cosmicSignals.push({
    id:        this._signalIdCounter++,
    starId:    star.id,
    civName:   star.civName,
    startTick: this.tick,
    decodedAt: this.tick + RADIO_DECODE_TICKS,
    message,
    fired:     false,
    decoded:   false,
  });
}

private updateCosmicSignals(): void {
  for (const sig of this.cosmicSignals) {
    if (!sig.fired) {
      sig.fired = true;
      this.onCosmicSignal?.(sig.civName, sig.starId);
    }
    if (!sig.decoded && this.tick >= sig.decodedAt) {
      sig.decoded = true;
      this.onSignalDecoded?.(sig.civName, sig.message);
    }
  }
  // Trim old decoded signals to prevent memory growth
  if (this.cosmicSignals.length > 50) {
    this.cosmicSignals = this.cosmicSignals.filter(s => !s.decoded).slice(-50);
  }
}
```
