# M16: AI Leaders, Dialogue & Memory — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Civilizations at Ancient tier (civLevel ≥ 1) generate named leaders with faction flags, ideology, and Gemini-powered dialogue that responds to in-game events and remembers past interactions.

**Architecture:** Leaders are stored on GameStateData (persisted in saves). BigBangEngine spawns/replaces leaders on civ tier changes and fires `onLeaderMessage` during major events. A new `LeaderDialogue.ts` builds prompts and calls the existing `GeminiService`/`FallbackNarrator` stack. A new modal in `index.html` shows the dialogue and response options.

**Tech Stack:** TypeScript, Canvas 2D, existing GeminiService (gemini-2.5-flash), FallbackNarrator, SeedRNG (mulberry32), pako save format.

```typescript
import { SeedRNG } from '../utils/SeedRNG';

export type LeaderRole = 'king' | 'scientist' | 'general' | 'prophet' | 'diplomat';
export type CommunicationStyle = 'formal_ruler' | 'religious_zealot' | 'military_commander' | 'nervous_diplomat' | 'scientific_thinker';
export type AttitudeTowardGod = 'loyal_worshipper' | 'fearful_servant' | 'skeptical_ruler' | 'manipulative_believer' | 'open_atheist';
export type PersonalityTrait = 'fanatic' | 'charismatic' | 'paranoid' | 'rational' | 'manipulative' | 'humble' | 'arrogant' | 'pacifist' | 'militaristic';
export type IdeologyLabel = 'Theocracy' | 'Republic' | 'Empire' | 'Confederation' | 'Technocracy' | 'Warband';

export interface Leader {
  id: string;
  starId: number;
  name: string;
  species: string;
  faction: string;
  role: LeaderRole;
  ideology: IdeologyLabel;
  intelligenceLevel: number;   // 1–10
  ambitionLevel: number;       // 1–10
  faithLevel: number;          // 1–10
  fearLevel: number;           // 1–10, dynamic
  personalityTraits: PersonalityTrait[];  // 2–3 traits
  communicationStyle: CommunicationStyle;
  attitudeTowardGod: AttitudeTowardGod;
  spawnedAtCivLevel: number;
}

export interface LeaderMemory {
  leaderId: string;
  entries: string[];  // max 10 short sentences
}

const FIRST_NAMES = [
  'Arath','Vorel','Syndra','Kalex','Theron','Zyra','Doven','Ixara',
  'Morven','Talis','Aeryn','Draxis','Selune','Koveth','Lyran','Nhara',
  'Exor','Velsha','Grond','Tavish','Quell','Ysara','Benth','Cirex',
];
const LAST_NAMES = [
  'of Kareth','the Undying','Vex','al-Sorn','Prime','of the Void',
  'Starkiller','the Ancient','Dawnbringer','Ironwill','Ashenmoor',
];
const ROLE_WEIGHTS: Record<string, LeaderRole[]> = {
  low_faith_low_hostility:   ['diplomat','scientist','king','king','diplomat'],
  low_faith_high_hostility:  ['general','general','king','general','diplomat'],
  high_faith_low_hostility:  ['prophet','king','prophet','diplomat','scientist'],
  high_faith_high_hostility: ['king','general','prophet','general','king'],
};
const STYLE_BY_ROLE: Record<LeaderRole, CommunicationStyle> = {
  king:      'formal_ruler',
  scientist: 'scientific_thinker',
  general:   'military_commander',
  prophet:   'religious_zealot',
  diplomat:  'nervous_diplomat',
};
const IDEOLOGY_BY_CIVLEVEL: IdeologyLabel[] = [
  'Warband',      // 0 (shouldn't spawn but guard)
  'Theocracy',    // 1 Ancient
  'Empire',       // 2 Medieval
  'Republic',     // 3 Industrial
  'Confederation',// 4 Atomic
  'Technocracy',  // 5 Space Age
  'Technocracy',  // 6+
];
const TRAITS: PersonalityTrait[] = ['fanatic','charismatic','paranoid','rational','manipulative','humble','arrogant','pacifist','militaristic'];

export function generateLeader(
  starId: number,
  civName: string,
  speciesName: string,
  civLevel: number,
  hostility: number,  // 0–20 from universe stats
  religionDevotion: number,  // 0–1
  rng: SeedRNG,
): Leader {
  const first = rng.pick(FIRST_NAMES);
  const last = rng.chance(0.5) ? ' ' + rng.pick(LAST_NAMES) : '';
  const name = first + last;

  const highFaith = religionDevotion > 0.4;
  const highHostility = hostility > 10;
  const roleKey = `${highFaith ? 'high' : 'low'}_faith_${highHostility ? 'high' : 'low'}_hostility`;
  const role = rng.pick(ROLE_WEIGHTS[roleKey]);

  const ideology = IDEOLOGY_BY_CIVLEVEL[Math.min(civLevel, IDEOLOGY_BY_CIVLEVEL.length - 1)];
  // Hostility override
  const finalIdeology: IdeologyLabel = highHostility && civLevel <= 2 ? 'Empire' : ideology;

  // Pick 2–3 traits, no repeats
  const shuffled = [...TRAITS].sort(() => rng.nextFloat(-1, 1));
  const traitCount = rng.chance(0.4) ? 3 : 2;
  const personalityTraits = shuffled.slice(0, traitCount) as PersonalityTrait[];

  // Starting attitude — religious civs start more reverent
  let attitudeTowardGod: AttitudeTowardGod = 'skeptical_ruler';
  if (religionDevotion > 0.6) attitudeTowardGod = 'loyal_worshipper';
  else if (religionDevotion > 0.3) attitudeTowardGod = 'fearful_servant';
  else if (highHostility) attitudeTowardGod = 'manipulative_believer';

  return {
    id: `leader_${starId}_${civLevel}_${Math.floor(rng.nextFloat(0, 99999))}`,
    starId,
    name,
    species: speciesName || civName,
    faction: civName,
    role,
    ideology: finalIdeology,
    intelligenceLevel: Math.ceil(rng.nextFloat(3, 10)),
    ambitionLevel:     Math.ceil(rng.nextFloat(2, 10)),
    faithLevel:        Math.round(religionDevotion * 10),
    fearLevel:         Math.ceil(rng.nextFloat(1, 5)),
    personalityTraits,
    communicationStyle: STYLE_BY_ROLE[role],
    attitudeTowardGod,
    spawnedAtCivLevel: civLevel,
  };
}

/**
 * Shift attitudeTowardGod based on a valence (+1 = positive, -1 = negative).
 * Returns new attitude string.
 */
export function shiftLeaderAttitude(current: AttitudeTowardGod, valence: number): AttitudeTowardGod {
  const SCALE: AttitudeTowardGod[] = [
    'open_atheist', 'skeptical_ruler', 'manipulative_believer',
    'fearful_servant', 'loyal_worshipper',
  ];
  const idx = SCALE.indexOf(current);
  const next = Math.max(0, Math.min(SCALE.length - 1, idx + valence));
  return SCALE[next];
}
```

### Step 2: Create `src/simulation/FactionFlag.ts`

```typescript
import { SeedRNG } from '../utils/SeedRNG';

export type FlagSymbol = 'star' | 'cross' | 'spiral' | 'flame' | 'eye' | 'wave';

export interface FactionFlag {
  primaryColor: string;
  secondaryColor: string;
  symbol: FlagSymbol;
}

const FLAG_COLORS = [
  '#cc4444','#cc7744','#cccc44','#44cc44','#44cccc',
  '#4488cc','#8844cc','#cc44cc','#ffffff','#aaaaaa',
];
const SYMBOLS: FlagSymbol[] = ['star','cross','spiral','flame','eye','wave'];

export function generateFactionFlag(rng: SeedRNG): FactionFlag {
  let primaryIdx = Math.floor(rng.nextFloat(0, FLAG_COLORS.length));
  let secondaryIdx = Math.floor(rng.nextFloat(0, FLAG_COLORS.length));
  if (secondaryIdx === primaryIdx) secondaryIdx = (primaryIdx + 1) % FLAG_COLORS.length;
  return {
    primaryColor:   FLAG_COLORS[primaryIdx],
    secondaryColor: FLAG_COLORS[secondaryIdx],
    symbol:         rng.pick(SYMBOLS),
  };
}

/**
 * Draw a small faction flag icon centered at (cx, cy) with given size (px).
 * Scale-gated — only call when zoom >= 0.6.
 */
export function drawFactionFlag(
  ctx: CanvasRenderingContext2D,
  flag: FactionFlag,
  cx: number,
  cy: number,
  size: number,
): void {
  ctx.save();
  // Background rectangle
  const w = size * 1.5;
  const h = size;
  ctx.fillStyle = flag.primaryColor;
  ctx.fillRect(cx - w / 2, cy - h / 2, w, h);

  // Border
  ctx.strokeStyle = flag.secondaryColor;
  ctx.lineWidth = size * 0.08;
  ctx.strokeRect(cx - w / 2, cy - h / 2, w, h);

  // Symbol in secondary color
  ctx.fillStyle = flag.secondaryColor;
  ctx.font = `${Math.round(size * 0.7)}px serif`;
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  const SYMBOL_CHARS: Record<FlagSymbol, string> = {
    star: '★', cross: '✚', spiral: '⊛', flame: '🔥', eye: '◉', wave: '≋',
  };
  // Use simple ASCII fallbacks to avoid emoji rendering issues on canvas
  const ASCII_FALLBACK: Record<FlagSymbol, string> = {
    star: '★', cross: '+', spiral: '@', flame: '^', eye: 'O', wave: '~',
  };
  try {
    ctx.fillText(SYMBOL_CHARS[flag.symbol] ?? ASCII_FALLBACK[flag.symbol], cx, cy);
  } catch {
    ctx.fillText(ASCII_FALLBACK[flag.symbol], cx, cy);
  }
  ctx.restore();
}
```

### Step 1: Create the file

```typescript
import type { Leader, LeaderMemory } from '../simulation/Leader';
import { GeminiService } from './GeminiService';
import { FallbackNarrator } from './FallbackNarrator';

interface DialogueResult {
  text: string;
  fromFallback: boolean;
}

function buildPrompt(
  leader: Leader,
  memories: LeaderMemory | undefined,
  eventContext: string,
): string {
  const memoryLines = memories?.entries.slice(-5) ?? [];
  const memoryBlock = memoryLines.length > 0
    ? `Relevant Memories:\n${memoryLines.map(e => `- ${e}`).join('\n')}`
    : 'No prior memories.';

  return `You are roleplaying a civilization leader speaking directly to a divine being (the player god).

Leader: ${leader.name}, ${leader.role} of ${leader.faction}
Species: ${leader.species}
Ideology: ${leader.ideology}
Personality: ${leader.personalityTraits.join(', ')}
Communication Style: ${leader.communicationStyle.replace(/_/g, ' ')}
Attitude Toward God: ${leader.attitudeTowardGod.replace(/_/g, ' ')}
Intelligence: ${leader.intelligenceLevel}/10, Ambition: ${leader.ambitionLevel}/10, Faith: ${leader.faithLevel}/10, Fear: ${leader.fearLevel}/10

${memoryBlock}

Current Situation: ${eventContext}

Instructions:
- Stay fully in character. Speak as if addressing a powerful god.
- Tone must reflect personality traits and communication style.
- Refer to past memories when relevant.
- Keep response concise (2–4 sentences).
- Avoid modern slang unless civilization is technologically advanced (Space Age+).
- Do not break character. No meta commentary.

Generate the message this leader would say to their god right now.`;
}

function fallbackDialogue(
  leader: Leader,
  eventContext: string,
  narrator: FallbackNarrator,
): string {
  // Use FallbackNarrator's civ milestone generator as approximation
  const base = narrator.generateCivilizationMilestone(leader.faction, eventContext);
  // Prepend leader name for context
  return `"${base}" — ${leader.name}, ${leader.role} of ${leader.faction}`;
}

/**
 * Generate a leader's spoken message for an event.
 * Uses GeminiService if available; falls back to FallbackNarrator.
 * Returns a promise so main.ts can await and display once ready.
 */
export async function generateLeaderDialogue(
  leader: Leader,
  eventContext: string,
  memories: LeaderMemory | undefined,
  geminiService: GeminiService | null,
  fallbackNarrator: FallbackNarrator,
): Promise<DialogueResult> {
  if (geminiService && !geminiService.offlineMode) {
    const prompt = buildPrompt(leader, memories, eventContext);
    try {
      // sendPlayerMessage accepts any string — we repurpose it for leader dialogue
      const text = await geminiService.sendPlayerMessage(prompt, '');
      return { text, fromFallback: false };
    } catch {
      // Fall through to fallback
    }
  }
  return {
    text: fallbackDialogue(leader, eventContext, fallbackNarrator),
    fromFallback: true,
  };
}

/**
 * Generate a short summary sentence to store as a leader memory entry.
 */
export async function generateMemoryEntry(
  leader: Leader,
  eventContext: string,
  geminiService: GeminiService | null,
  fallbackNarrator: FallbackNarrator,
): Promise<string> {
  if (geminiService && !geminiService.offlineMode) {
    const prompt = `Summarize in one short sentence (max 15 words) what ${leader.name} of ${leader.faction} will remember about this event: "${eventContext}". Write in first-person past tense as the leader.`;
    try {
      const text = await geminiService.sendPlayerMessage(prompt, '');
      return text.slice(0, 120);
    } catch { /* fallback */ }
  }
  return `${eventContext} — remembered by ${leader.name}.`;
}
```

### Step 4: Add helper functions

```typescript
function appendLeaderMemory(leaderId: string, entry: string): void {
  let mem = gameState.leaderMemories.find(m => m.leaderId === leaderId);
  if (!mem) {
    mem = { leaderId, entries: [] };
    gameState.leaderMemories.push(mem);
  }
  mem.entries.push(entry);
  if (mem.entries.length > 10) mem.entries.shift();
}

function handleLeaderResponse(cost: number, attitudeValence: number, action: string, label: string): void {
  if (!currentDialogueLeader) return;
  if (gameState.divinePoints < cost) {
    addMsg('Not enough Divine Points.', 'system');
    return;
  }
  if (cost > 0) awardDP(-cost, `Leader response: ${label}`);

  const leader = currentDialogueLeader;
  // Shift attitude
  if (attitudeValence !== 0) {
    leader.attitudeTowardGod = shiftLeaderAttitude(leader.attitudeTowardGod, attitudeValence);
    document.getElementById('leader-dlg-attitude')!.textContent =
      `Attitude: ${leader.attitudeTowardGod.replace(/_/g, ' ')}`;
  }

  if (action === 'bless') {
    const star = engine.getStarById(leader.starId);
    if (star) star.religionDevotion = Math.min(1, star.religionDevotion + 0.2);
    addMsg(`You bestowed a blessing upon ${leader.name}'s people.`, 'god');
    appendLeaderMemory(leader.id, 'The god granted us a blessing of devotion.');
  } else if (action === 'omen') {
    addMsg(`You sent an omen to ${leader.name}. Fear spreads among the faithful.`, 'god');
    leader.fearLevel = Math.min(10, leader.fearLevel + 2);
    appendLeaderMemory(leader.id, 'A divine omen filled our hearts with dread.');
  } else {
    addMsg(`You watched ${leader.name} in silence.`, 'narration');
  }

  closeLeaderDialogue();
}

function closeLeaderDialogue(): void {
  document.getElementById('leader-dialogue-overlay')!.classList.remove('active');
  currentDialogueLeader = null;
}
```
